-- (WHERE + ORDER BY)

-- Display all active employees.
select * from employees where emp_status="Active";
-- Display all female employees.
select * from employees where gender="female" ;
-- Display employees hired after 2026-03-01.
select * from employees where hire_date > '2026-03-01';
-- Display employees working in the IT department.
select * from department where dept_name="Information Technology";
-- Display employees ordered by hire date.
select * from employees order by hire_date asc;
-- Display salaries in descending order.
select * from payroll order by basic_salary desc;


-- (Aggregate Functions)

-- How many employees are there in the company?
select count(*) from employees;
-- What is the total basic salary paid by the company?
select sum(basic_salary) from payroll ;
-- What is the average basic salary?
select avg(basic_salary) from payroll;
-- What is the highest salary?
select max(basic_salary) from payroll;
-- What is the lowest salary?
select min(basic_salary) from payroll;

-- GROUP BY

-- How many employees are there in each department?
 select d.dept_name,
count(e.emp_id) as no_of_employees
from department d
join employees e
on d.dept_id = e.dept_id
group by d.dept_name;

-- How many employees are there in each gender category?
select gender,count(gender)
 from employees
 group by  gender;

-- What is the total salary paid to employees in each department?
select d.dept_name,
sum(p.basic_salary) as total_salary
 from payroll as p
 join employees as e
 on p.emp_id = e.emp_id
 join department as d
 on e.dept_id=d.dept_id
 group by  d.dept_name;


-- What is the average salary in each department?
select  emp_id,avg(basic_salary) as average_salary
 from payroll
 join department
 group by emp_id;


-- Which department has the highest total salary?
select max(basic_salary) as  highest_salary,dept_name
from payroll as p
join employees as e
on p.emp_id=e.emp_id
join department as d
on e.dept_id=d.dept_id
group by dept_name 
order by highest_salary desc
limit 1 ;

-- JOINS

-- . Display employee name and department name
select d.dept_name,e.emp_name
from department as d
left join employees as e
on d.dept_id=e.dept_id;

-- Display employee name and salary
select e.emp_name,p.basic_salary
from employees as e
left join payroll as p
on e.emp_id=p.emp_id;

-- Display employee name, department name and salary
select e.emp_name,p.basic_salary,d.dept_name
from department as d
left join employees as e
on d.dept_id=e.dept_id
left join payroll as p
on e.emp_id = p.emp_id;

-- Display employee name and project title

on e.emp_id=ep.emp_id
left join project as p
on p.project_id=ep.project_id;

-- Display employee name, department name, project title and role
select d.dept_name,e.emp_name,p.project_title,ep.project_role
from department as d
left join employees as e
on d.dept_id=e.dept_id
left join employee_project as ep
on e.emp_id=ep.emp_id

-- HAVING 
-- Show departments having more than 1 employee
select * from employees;
select * from department;
select count(*)as emp_count
from department as d
left join employees AS e
on d.dept_id=e.dept_id
group by dept_name
having 1_emp > 1;

-- Show departments whose total salary is greater than 40000
select max(p.basic_salary) AS total_salary,d.dept_name  
from payroll as p
left join employees as e
on p.emp_id=e.emp_id
left join department as d
on e.dept_id=d.dept_id
group by dept_name
having total_salary > 40000 
order by total_salary desc
 limit 1 ;

-- Show departments whose average salary is greater than 30000
select avg(p.basic_salary) AS average_salary,d.dept_name  
from payroll as p
left join employees as e
on p.emp_id=e.emp_id
left join department as d
on e.dept_id=d.dept_id
group by dept_name
having average_salary > 30000 
order by average_salary desc;

-- SUBQUERIES (EXISTS, ANY, ALL, IN)
-- Find the employee with the highest salary.
select a.basic_salary ,e.emp_name
from payroll as a
left join employees as e
on a.emp_id = e.emp_id
where a.basic_salary =(
select max(basic_salary) as highest_salary
 from payroll
 limit 1
 );

-- Find the employee with the lowest salary.
select p.basic_salary,e.emp_name
from payroll as p
left join employees as e
on p.emp_id = e.emp_id
where p.basic_salary=(
select min(basic_salary) as minimum_salary
from payroll
);

-- Find employees earning above average salary.
select e.emp_name, p.basic_salary
from payroll as p
left join employees as e
on p.emp_id = e.emp_id
where p.basic_salary >(
select avg(basic_salary) as average_salary
from payroll
);


-- Find employees earning below average salary.
select e.emp_name, p.basic_salary
from payroll as p
left join employees as e
on p.emp_id = e.emp_id
where basic_salary <(
select avg(basic_salary) as lowest_salary
from payroll);
